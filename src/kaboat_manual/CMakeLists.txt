cmake_minimum_required(VERSION 3.8)
project(kaboat_manual)
find_package(ament_cmake REQUIRED)
install(PROGRAMS manual.py DESTINATION lib/${PROJECT_NAME} RENAME manual)

  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(manual_launcher test_manual_launcher.py)
endif()
ament_package()
